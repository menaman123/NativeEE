# tests/ipu -- Phase 7 IPU suites (PHASE7_PLAN section 2.5 / 4.2). Stage 7C: the
# IpuDevice register/FIFO/command contract and the MpegDecode arithmetic. Stage 7D:
# the hle pipeline pieces (PssDemux, FrameWriter, FmvPacer, FmvAudioCapture); 7E
# the per-picture oracle.
function(ps2r_ipu_test name ctest_name)
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE ps2r_ipu_model)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror)
    add_test(NAME ${ctest_name} COMMAND ${name} ${ARGN})
    set_tests_properties(${ctest_name} PROPERTIES TIMEOUT 120)
endfunction()

# Stage 7E. fmv_picture_oracle: the per-picture golden tool (PHASE7_PLAN section 4.3).
# ctest FmvPictureOracle decodes the two intro movies under ${PS2R_OG_ROOT} and asserts
# the OLD golden manifest fixtures/coldboot_pictures.sha256 is a contiguous window of
# movie 1's decoded pictures reaching its last picture, followed by the same for movie 2
# (P7-GOLDEN-WINDOW); registered as a SKIP echo when the OLD tree or the fixture is absent.
add_executable(fmv_picture_oracle fmv_picture_oracle.cpp file_fmv_sector_source.cpp)
target_link_libraries(fmv_picture_oracle PRIVATE ps2r_ipu)
target_compile_options(fmv_picture_oracle PRIVATE -Wall -Wextra -Werror)
set(_p7_golden ${CMAKE_CURRENT_SOURCE_DIR}/fixtures/coldboot_pictures.sha256)
if(PS2R_OG_ROOT AND EXISTS "${PS2R_OG_ROOT}/game_files" AND EXISTS "${_p7_golden}")
    add_test(NAME FmvPictureOracle COMMAND fmv_picture_oracle --og-root "${PS2R_OG_ROOT}"
        --movie FMV/VIVENDI.PSS --movie FMV/TTIDENT.PSS --manifest "${_p7_golden}")
else()
    add_test(NAME FmvPictureOracle COMMAND ${CMAKE_COMMAND} -E echo
        "SKIP: FmvPictureOracle (OLD tree or fixtures/coldboot_pictures.sha256 absent)")
endif()
set_tests_properties(FmvPictureOracle PROPERTIES TIMEOUT 600)
